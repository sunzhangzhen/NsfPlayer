cmake_minimum_required(VERSION 3.22.1)
project(nsfcore CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB GME_SRCS ${CMAKE_CURRENT_SOURCE_DIR}/gme/*.cpp)
add_library(gme STATIC ${GME_SRCS})
target_include_directories(gme PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/gme ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(gme PRIVATE BLARGG_BUILD_DLL)
add_library(nsfcore SHARED native-lib.cpp)
target_link_libraries(nsfcore gme android log)
